package scaffold

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// ErrProjectExists indicates apitest.yaml already exists in the target directory.
var ErrProjectExists = errors.New("project already exists: apitest.yaml found in target directory")

// Options configures project initialization.
type Options struct {
	Dir         string // Target directory; empty defaults to "."
	ProjectName string // Override project name; empty defaults to directory basename
}

// Init creates a new apitest project in the specified directory.
// Creates apitest.yaml, .gitignore (or appends), .env.example,
// environments/dev.yaml, and collections/sample.yaml.
// Returns ErrProjectExists if apitest.yaml or apitest.yml already exists.
func Init(opts Options) error {
	dir := opts.Dir
	if dir == "" {
		dir = "."
	}

	// Detect existing project
	for _, name := range []string{"apitest.yaml", "apitest.yml"} {
		if _, err := os.Stat(filepath.Join(dir, name)); err == nil {
			return fmt.Errorf("%w", ErrProjectExists)
		}
	}

	projectName := opts.ProjectName
	if projectName == "" {
		abs, err := filepath.Abs(dir)
		if err != nil {
			return fmt.Errorf("resolving directory: %w", err)
		}
		projectName = filepath.Base(abs)
	}

	// Create subdirectories
	for _, d := range []string{"environments", "collections"} {
		if err := os.MkdirAll(filepath.Join(dir, d), 0o750); err != nil {
			return fmt.Errorf("creating %s directory: %w", d, err)
		}
	}

	if err := writeFile(filepath.Join(dir, "apitest.yaml"), apitestYAML(projectName)); err != nil {
		return err
	}
	if err := ensureGitignore(filepath.Join(dir, ".gitignore")); err != nil {
		return err
	}
	if err := writeFile(filepath.Join(dir, ".env.example"), envExample()); err != nil {
		return err
	}
	if err := writeFile(filepath.Join(dir, "environments", "dev.yaml"), devEnvironment()); err != nil {
		return err
	}
	if err := writeFile(filepath.Join(dir, "collections", "sample.yaml"), sampleCollection()); err != nil {
		return err
	}

	return nil
}

func writeFile(path, content string) error {
	if err := os.WriteFile(path, []byte(content), 0o644); err != nil { //nolint:gosec
		return fmt.Errorf("writing %s: %w", filepath.Base(path), err)
	}
	return nil
}

func ensureGitignore(path string) error {
	existing, err := os.ReadFile(path)
	if err != nil && !errors.Is(err, os.ErrNotExist) {
		return fmt.Errorf("reading .gitignore: %w", err)
	}

	if err == nil {
		// File exists — check if .env line already present
		for _, line := range strings.Split(string(existing), "\n") {
			if strings.TrimSpace(line) == ".env" {
				return nil
			}
		}
		// Append .env
		content := string(existing)
		if !strings.HasSuffix(content, "\n") {
			content += "\n"
		}
		return os.WriteFile(path, []byte(content+".env\n"), 0o644) //nolint:gosec
	}

	return writeFile(path, ".env\n")
}

func apitestYAML(projectName string) string {
	return fmt.Sprintf("project_name: %s\nvariables:\n  base_url: \"https://httpbin.org\"\n", projectName)
}

func envExample() string {
	return "# Local secrets for this project.\n" +
		"# Copy this file to .env and fill in real values.\n" +
		"# NEVER commit .env to version control.\n" +
		"#\n" +
		"# API_KEY=your-api-key-here\n" +
		"# SECRET_TOKEN=your-secret-here\n"
}

func devEnvironment() string {
	return "variables:\n  base_url: \"https://httpbin.org\"\n"
}

func sampleCollection() string {
	return "name: Sample Collection\n" +
		"description: A sample collection generated by apitest init\n\n" +
		"requests:\n" +
		"  - name: Hello World\n" +
		"    request:\n" +
		"      method: GET\n" +
		"      url: \"{{base_url}}/get\"\n" +
		"    assertions:\n" +
		"      status: 200\n"
}
```

func TestInit(t *testing.T) {
	tests := []struct {
		name         string
		setup        func(t *testing.T, dir string)
		opts         func(dir string) Options
		wantErr      error
		wantFiles    []string
		checkContent map[string]string // file path -> required substring
	}{

			name: "creates all files in empty directory",
			wantFiles: []string{
				"apitest.yaml",
				".gitignore",
				".env.example",
				"environments/dev.yaml",
				"collections/sample.yaml",
			},
		},
		{
			name: "apitest.yaml contains project name from directory basename",
			checkContent: map[string]string{
				"apitest.yaml": "project_name:",
			},
		},
		{
			name: "apitest.yaml uses custom project name when provided",
			opts: func(dir string) Options { return Options{Dir: dir, ProjectName: "MyAPI"} },
			checkContent: map[string]string{
				"apitest.yaml": "project_name: MyAPI",
			},
		},
		{
			name: "gitignore contains .env entry",
			checkContent: map[string]string{
				".gitignore": ".env",
			},
		},
		{
			name: "env.example documents variables without real values",
			checkContent: map[string]string{
				".env.example": "NEVER commit .env",
			},
		},
		{
			name: "sample collection is valid yaml with base_url variable",
			checkContent: map[string]string{
				"collections/sample.yaml": "{{base_url}}",
			},
		},
		{
			name: "error when apitest.yaml already exists",
			setup: func(t *testing.T, dir string) {
				_ = os.WriteFile(filepath.Join(dir, "apitest.yaml"), []byte("project_name: Existing\n"), 0o600)
			},
			wantErr: ErrProjectExists,
		},
		{
			name: "error when apitest.yml already exists",
			setup: func(t *testing.T, dir string) {
				_ = os.WriteFile(filepath.Join(dir, "apitest.yml"), []byte("project_name: Existing\n"), 0o600)
			},
			wantErr: ErrProjectExists,
		},
		{
			name: "appends .env to existing gitignore without duplication",
			setup: func(t *testing.T, dir string) {
				_ = os.WriteFile(filepath.Join(dir, ".gitignore"), []byte("node_modules/\n"), 0o600)
			},
			checkContent: map[string]string{
				".gitignore": "node_modules/",
			},
		},
		{
			name: "does not duplicate .env when gitignore already has it",
			setup: func(t *testing.T, dir string) {
				_ = os.WriteFile(filepath.Join(dir, ".gitignore"), []byte(".env\n"), 0o600)
			},
		},
		{
			name: "env.example contains no real secret values",
			checkContent: map[string]string{
				".env.example": "your-api-key-here",
			},
		},
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			dir := t.TempDir()
			if tc.setup != nil {
				tc.setup(t, dir)
			}
			opts := Options{Dir: dir}
			if tc.opts != nil {
				opts = tc.opts(dir)
			}

			err := Init(opts)

			if tc.wantErr != nil {
				if !errors.Is(err, tc.wantErr) {
					t.Fatalf("want error %v, got %v", tc.wantErr, err)
				}
				return
			}
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}

			for _, f := range tc.wantFiles {
				path := filepath.Join(dir, filepath.FromSlash(f))
				if _, err := os.Stat(path); err != nil {
					t.Errorf("expected file %s to exist: %v", f, err)
				}
			}

			for rel, substr := range tc.checkContent {
				path := filepath.Join(dir, filepath.FromSlash(rel))
				data, err := os.ReadFile(path)
				if err != nil {
					t.Fatalf("read %s: %v", rel, err)
				}
				if !strings.Contains(string(data), substr) {
					t.Errorf("file %s: expected to contain %q\ngot:\n%s", rel, substr, data)
				}
			}
		})
	}
}

func initCmd(args []string) int {
	dir := "."
	if len(args) > 0 {
		dir = args[0]
	}

	absDir, err := filepath.Abs(dir)
	if err != nil {
		_, _ = fmt.Fprintf(os.Stderr, "Error: %v\n", err)
		return 1
	}
	projectName := filepath.Base(absDir)

	if err := scaffold.Init(scaffold.Options{
		Dir:         dir,
		ProjectName: projectName,
	}); err != nil {
		_, _ = fmt.Fprintf(os.Stderr, "Error: %v\n", err)
		return 1
	}

	fmt.Println("Project initialized successfully!")
	fmt.Println()
	fmt.Println("Created:")
	fmt.Println("  apitest.yaml")
	fmt.Println("  .gitignore")
	fmt.Println("  .env.example")
	fmt.Println("  environments/dev.yaml")
	fmt.Println("  collections/sample.yaml")
	fmt.Println()
	fmt.Println("Next steps:")
	fmt.Println("  apitest run collections/sample.yaml")
	return 0
}
```

// runBinaryInDir runs the binary with args in a specific working directory.
func runBinaryInDir(t *testing.T, binary, dir string, args ...string) (stdout, stderr string, exitCode int) {
	t.Helper()
	cmd := exec.Command(binary, args...)
	cmd.Dir = dir
	var outBuf, errBuf strings.Builder
	cmd.Stdout = &outBuf
	cmd.Stderr = &errBuf
	err := cmd.Run()
	exitCode = 0
	if exitErr, ok := err.(*exec.ExitError); ok {
		exitCode = exitErr.ExitCode()
	} else if err != nil {
		t.Fatalf("unexpected error running binary: %v", err)
	}
	return outBuf.String(), errBuf.String(), exitCode
}

func TestInitThenRun_integration(t *testing.T) {
	if testing.Short() {
		t.Skip("skipping integration test in short mode")
	}
	binary := buildBinary(t)
	dir := t.TempDir()

	// Step 1: init
	stdout, stderr, code := runBinaryInDir(t, binary, dir, "init")
	if code != 0 {
		t.Fatalf("init failed (code=%d): stdout=%q stderr=%q", code, stdout, stderr)
	}
	if !strings.Contains(stdout, "Project initialized successfully") {
		t.Errorf("expected success message, got: %q", stdout)
	}

	// Step 2: run the generated sample collection
	stdout, stderr, code = runBinaryInDir(t, binary, dir, "run", "collections/sample.yaml")
	if code != 0 {
		t.Fatalf("run sample collection failed (code=%d): stdout=%q stderr=%q", code, stdout, stderr)
	}
}

## Risks and Edge Cases

- **Risk:** `.gitignore` already has `.env` → **Mitigation:** `ensureGitignore` checks line-by-line; no duplicate entry.
- **Risk:** `.gitignore` has `.env.local` but not `.env` → **Mitigation:** exact `strings.TrimSpace(line) == ".env"` check; no confusion.
- **Risk:** Sample collection uses `{{base_url}}` — must resolve at runtime → **Mitigation:** Generated `apitest.yaml` defines `base_url` as a project variable; `LoadProjectConfig` walk-up finds it when running from the project directory.
- **Risk:** Directory name has YAML-special characters → **Mitigation:** `apitestYAML` uses `fmt.Sprintf` with an unquoted value; simple names (the common case) work fine. V1 accepts this limitation.
- **Edge case:** `apitest.yml` variant already exists → detected in the dual-check loop, returns `ErrProjectExists`.
- **Edge case:** `init` called with no args → defaults to `"."`, derives project name from `filepath.Base(filepath.Abs("."))`.
- **Edge case:** Root directory passed as `dir` → `filepath.Base("/")` returns `"/"` but this is an extreme edge case, acceptable for V1.
- **Edge case:** No write permissions → `os.WriteFile` / `os.MkdirAll` surface the OS error wrapped with context.
